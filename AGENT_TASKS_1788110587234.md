```kotlin
/**
 * Generate a session PIN, store its hash, expose plain-text via [sessionPin] flow.
 * Call this before enter() — the UI shows the PIN then calls enter().
 */
fun preparePin(): String {
    val plain = PinPolicy.generate()
    Database.setSetting(LAUNCHER_PIN_KEY, sha256(plain))
    _sessionPin.value = plain
    return plain
}

/** Verify a PIN attempt against the stored session PIN hash. */
fun verifyPin(raw: String): Boolean {
    val stored = Database.getSetting(LAUNCHER_PIN_KEY) ?: return true
    if (stored.isBlank()) return true
    return sha256(raw) == stored
}

/**
 * Called by ProcessMonitor's foreground callback on every foreground change.
 * Raises or lowers the launcher overlay based on whether the frontmost process
 * is in the allowed set.
 */
fun onForegroundChanged(processName: String) {
    if (!_isActive.value || _breakActive.value) return
    val allowed = _sessionApps.value.map { it.processName.lowercase() }.toSet()
    _overlayVisible.value = processName.lowercase() !in allowed
}

private fun sha256(input: String): String {
    val bytes = java.security.MessageDigest.getInstance("SHA-256")
        .digest(input.toByteArray(Charsets.UTF_8))
    return bytes.joinToString("") { "%02x".format(it) }
}
```

## T-04 — Create `LauncherWindow.kt`

**File (new):** `src/main/kotlin/com/focusflow/ui/launcher/LauncherWindow.kt`

```kotlin
package com.focusflow.ui.launcher

import androidx.compose.runtime.*
import androidx.compose.ui.unit.DpSize
import androidx.compose.ui.unit.dp
import androidx.compose.ui.window.*
import com.focusflow.enforcement.User32Extra
import com.focusflow.enforcement.isWindows
import com.focusflow.services.FocusLauncherService
import com.sun.jna.Native
import java.awt.GraphicsEnvironment
import java.awt.Rectangle
import javax.swing.JFrame

/**
 * Renders one undecorated fullscreen window per screen device when the launcher
 * session is active.  The primary monitor shows the full launcher UI; secondary
 * monitors show a simple lock screen.
 *
 * Window management:
 *   - undecorated = true  → no title bar, no min/max/close buttons
 *   - alwaysOnTop = true  → AWT-level, raised on creation
 *   - JNA SetWindowPos(HWND_TOPMOST) applied immediately after creation for
 *     Win32-level topmost (stronger than AWT alwaysOnTop)
 *   - When overlayVisible = false (allowed app in foreground):
 *       SetWindowPos(HWND_NOTOPMOST, SWP_NOACTIVATE) — lowers Z-order so the
 *       allowed app can come to the front without the launcher hiding it
 *   - When overlayVisible = true again:
 *       SetWindowPos(HWND_TOPMOST) — reclaims topmost Z-order
 *
 * Lifecycle: composed when isActive=true, removed from tree when isActive=false.
 * Compose Desktop creates/destroys the OS window accordingly.
 */
@Composable
fun LauncherWindowHost() {
    val isActive by FocusLauncherService.isActive.collectAsState()
    if (!isActive) return

    val screens = remember {
        GraphicsEnvironment.getLocalGraphicsEnvironment().screenDevices.toList()
    }
    val primaryBounds = remember {
        GraphicsEnvironment.getLocalGraphicsEnvironment()
            .defaultScreenDevice.defaultConfiguration.bounds
    }

    screens.forEach { screen ->
        val bounds    = screen.defaultConfiguration.bounds
        val isPrimary = bounds == primaryBounds
        key(bounds.x, bounds.y) {          // stable key per monitor position
            LauncherWindowForScreen(bounds = bounds, isPrimary = isPrimary)
        }
    }
}

@Composable
private fun LauncherWindowForScreen(bounds: Rectangle, isPrimary: Boolean) {
    val overlayVisible by FocusLauncherService.overlayVisible.collectAsState()
    val breakActive    by FocusLauncherService.breakActive.collectAsState()

    // Use Floating placement so we control exact position/size manually.
    // WindowPlacement.Fullscreen has multi-monitor quirks on Windows.
    val windowState = rememberWindowState(
        placement = WindowPlacement.Floating,
        position  = WindowPosition(bounds.x.dp, bounds.y.dp),
        size      = DpSize(bounds.width.dp, bounds.height.dp)
    )

    Window(
        onCloseRequest = { /* block all close attempts */ },
        state          = windowState,
        undecorated    = true,
        alwaysOnTop    = true,   // AWT level — SetWindowPos below makes it Win32-level
        resizable      = false,
        focusable      = true,
        title          = "FocusFlow Launcher"
    ) {
        // Block the close button at AWT/Swing level too.
        LaunchedEffect(Unit) {
            (window as? JFrame)?.defaultCloseOperation = JFrame.DO_NOTHING_ON_CLOSE
        }

        // Apply Win32 TOPMOST immediately after creation, then track overlayVisible.
        // During a break: lower the window so the user has real desktop access.
        val shouldBeTopmost = overlayVisible && !breakActive
        LaunchedEffect(shouldBeTopmost) {
            if (!isWindows) return@LaunchedEffect
            val hwnd = try {
                val ptr = Native.getComponentPointer(window)
                if (ptr != null) com.sun.jna.platform.win32.WinDef.HWND(ptr) else null
            } catch (_: Throwable) { null } ?: return@LaunchedEffect

            if (shouldBeTopmost) {
                User32Extra.INSTANCE.SetWindowPos(
                    hwnd, User32Extra.HWND_TOPMOST,
                    bounds.x, bounds.y, bounds.width, bounds.height,
                    User32Extra.SWP_SHOWWINDOW
                )
            } else {
                User32Extra.INSTANCE.SetWindowPos(
                    hwnd, User32Extra.HWND_NOTOPMOST,
                    bounds.x, bounds.y, bounds.width, bounds.height,
                    User32Extra.SWP_NOACTIVATE or User32Extra.SWP_NOMOVE or User32Extra.SWP_NOSIZE
                )
            }
        }

        if (isPrimary) {
            LauncherContent()
        } else {
            SecondaryScreenLock()
        }
    }
}
```

**Done when:** File compiles. Running the app should not crash on startup (the host renders
nothing while `isActive=false`).

```kotlin
@Composable
private fun LauncherTopBar(isHardLocked: Boolean) {
    var clock    by remember { mutableStateOf("") }

@Composable
private fun AppTile(app: com.focusflow.services.FocusLauncherApp) {
    val scope = rememberCoroutineScope()

    Box(
        modifier = Modifier
            .aspectRatio(1f)
            .clip(RoundedCornerShape(18.dp))

// ── Secondary screen lock ────────────────────────────────────────────────────

@Composable
fun SecondaryScreenLock() {
    var timer by remember { mutableStateOf("") }

    LaunchedEffect(Unit) {
        while (true) {
            val rem = FocusLauncherService.remainingSeconds()
            timer = if (rem > 0) "%02d:%02d".format(rem / 60, rem % 60) else ""
            delay(1_000)
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color(0xFF0C0B14)),
        contentAlignment = Alignment.Center
    ) {
        Column(
            horizontalAlignment = Alignment.CenterHorizontally,
            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {
            Icon(Icons.Default.Lock, null, tint = Purple80.copy(alpha = 0.4f),
                modifier = Modifier.size(36.dp))
            Text("Focus session in progress",
                color = OnSurface2, fontSize = 14.sp)
            if (timer.isNotEmpty())
                Text(timer, color = Purple80, fontSize = 20.sp, fontWeight = FontWeight.SemiBold)
        }
    }
}
```
